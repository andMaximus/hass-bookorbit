#include <tunables/global>

profile bookorbit flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # BookOrbit needs no elevated capabilities: no CIFS mounting (network shares come
  # from Home Assistant's own storage feature), no kernel modules, no raw sockets.
  # These are what s6, the entrypoint's ownership fixes and su-exec/setuidgid need.
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability net_bind_service,
  capability sys_chroot,

  network inet stream,
  network inet6 stream,
  network unix stream,
  network unix dgram,
  network netlink raw,

  # S6 supervision tree
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /usr/libexec/** ix,
  /command/** ix,
  /package/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Shells and the add-on's own scripts
  /bin/bash ix,
  /bin/sh ix,
  /usr/lib/bashio/** r,
  /etc/s6-overlay/** rix,

  # Application runtime
  /app/** r,
  /app/bin/** ix,
  /app/entrypoint.sh rix,
  /opt/bookorbit-python/** rix,
  /usr/lib/postgresql*/** rm,
  /usr/share/postgresql*/** r,

  # Add-on data: the Postgres cluster, app data and generated secrets
  /data/** rwlk,

  # Libraries. /media and /share cover Home Assistant network storage mounts;
  # /config is the add-on's own config folder.
  /media/** rwlk,
  /share/** rwlk,
  /config/** rwlk,
  /ssl/** r,
  /tmp/** rwlk,

  # The upstream entrypoint sizes the Node heap from the cgroup memory limit.
  /sys/fs/cgroup/memory.max r,
  /sys/fs/cgroup/memory/memory.limit_in_bytes r,
  /sys/fs/cgroup/** r,
  /proc/** r,
  /proc/*/oom_score_adj w,

  /etc/{passwd,group,hosts,resolv.conf,localtime,nsswitch.conf} r,
  /etc/ssl/** r,
  /usr/share/zoneinfo/** r,
}
