#include <tunables/global>

profile bookorbit flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # --- Capabilities -----------------------------------------------------------
  # BookOrbit needs nothing elevated. Network shares come from Home Assistant's
  # storage feature rather than being mounted in here, so there is no sys_admin,
  # no module loading, no raw sockets, no ptrace.
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability net_bind_service,
  capability sys_chroot,

  network inet stream,
  network inet6 stream,
  network unix stream,
  network unix dgram,
  network netlink raw,

  # s6 supervises the tree with signals, and PostgreSQL's postmaster signals its
  # children; constraining these buys nothing and breaks shutdown.
  signal,

  # --- Reads ------------------------------------------------------------------
  # Read is granted broadly and deliberately. Everything readable here is either
  # the add-on's own image or a volume Home Assistant mapped in on purpose, and
  # scripts must be readable by their interpreter: /init is `#!/bin/sh`, so `ix`
  # without `r` fails with "/bin/sh: can't open '/init': Permission denied".
  # The confinement that matters for a container is below - what it may execute,
  # write, and which capabilities it holds.
  /** r,
  /usr/lib/** rm,
  /lib/** rm,
  /usr/lib/postgresql*/** rm,

  # --- Execution --------------------------------------------------------------
  /init rix,
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/local/bin/** rix,
  /usr/libexec/** rix,
  /command/** rix,
  /package/** rix,
  /etc/s6-overlay/** rix,
  /app/entrypoint.sh rix,
  /app/bin/** rix,
  /opt/bookorbit-python/** rix,

  # --- Writes -----------------------------------------------------------------
  # The add-on's own data (PostgreSQL cluster, covers, book bucket, secrets).
  /data/** rwlk,

  # Libraries. /media and /share cover Home Assistant network storage mounts.
  /media/** rwlk,
  /share/** rwlk,
  /config/** rwlk,

  /tmp/** rwlk,
  /run/** rwlk,
  /dev/tty rw,
  /dev/null rw,
  /proc/*/oom_score_adj w,
}
